Fraud flags

A payment processor screens card transactions. Several signals can apply at once and an analyst wants all of them, not just the first: each rule is a flag with risk points.

The problem

Fraud flags, in words.

  • large_amount: more than three times the customer's average (a field compared with an expression of another field);
  • country_mismatch: the IP's country differs from the card's (two fields compared);
  • velocity: five or more transactions in the last hour;
  • new_device_large: a new device and more than 500;
  • disposable_email: an address at a throwaway mail service (a regular expression);
  • night_time: between midnight and 5 a.m., local time.

rule_order selects every rule that fired, in rule order.

fraud.rsRust core · runs
//! Fraud flags: every rule that fires, under the RULE_ORDER hit policy.
use cleanrules::Value;
use cleanrules::ast::Closed;
use cleanrules::build::{Build, constant, field};
use cleanrules::format::{ValueContext, read_value, write_value};
use cleanrules::json::{Limits, parse, to_string};
use cleanrules::rules::{Config, HitPolicy, Rule, RuleSet};

type Result<T> = std::result::Result<T, Box<dyn std::error::Error>>;

fn rule(name: &str, when: Build, then: Vec<(&str, Build)>) -> Result<Rule> {
    let then = then
        .into_iter()
        .map(|(key, value)| Ok((key.to_owned(), value.build()?)))
        .collect::<Result<Vec<_>>>()?;
    Ok(Rule::with_payload(name, when.build()?, then)?)
}

fn input(json: &str) -> Result<Value> {
    Ok(read_value(
        &parse(json, &Limits::default())?,
        "",
        ValueContext::Data,
    )?)
}

fn flags() -> Result<RuleSet> {
    let risk = |points: i64| vec![("risk", constant(points))];
    let rules = vec![
        rule(
            "large_amount",
            field("amount").gt(field("avg_amount") * 3),
            risk(40),
        )?,
        rule(
            "country_mismatch",
            field("ip_country").ne(field("card_country")),
            risk(25),
        )?,
        rule("velocity", field("tx_last_hour").ge(5), risk(30))?,
        rule(
            "new_device_large",
            field("new_device") & field("amount").gt(500),
            risk(20),
        )?,
        rule(
            "disposable_email",
            field("email").matches(r"@(mailinator|guerrillamail|10minutemail)\.com$"),
            risk(35),
        )?,
        rule(
            "night_time",
            field("local_hour").between_closed(0, 5, Closed::Both),
            risk(10),
        )?,
    ];
    let config = Config {
        hit_policy: HitPolicy::RuleOrder,
        ..Config::default()
    };
    Ok(RuleSet::new(rules, config)?)
}

fn main() -> Result<()> {
    let flags = flags()?;
    let tx = input(
        r#"{"amount": 1890.0, "avg_amount": 240.0, "ip_country": "NG", "card_country": "ES",
            "tx_last_hour": 2, "new_device": true, "email": "j.doe@mailinator.com",
            "local_hour": 14}"#,
    )?;
    let decision = flags.evaluate(&tx)?;
    for result in decision.selected() {
        let (_, risk) = &result.output()[0];
        println!(
            "{:<17} risk {}",
            result.rule.name(),
            to_string(&write_value(risk))
        );
    }
    match decision.output() {
        Ok(_) => println!("one output"),
        Err(error) => println!("output(): {} ({})", error, error.code()),
    }
    Ok(())
}

// Design preview: the Java binding does not exist yet (phase G).
import io.github.cleanrules.*;

import static io.github.cleanrules.Conditions.*;

public class Fraud {
    record Transaction(double amount, double avg_amount, String ip_country, String card_country,
                       int tx_last_hour, boolean new_device, String email, int local_hour) {}

    public static void main(String[] args) {
        var flags = RuleSet.builder()
            .hitPolicy(HitPolicy.RULE_ORDER)
            .add(Rule.named("large_amount").when(field("amount").gt(field("avg_amount").times(3)))
                .then("risk", 40))
            .add(Rule.named("country_mismatch").when(field("ip_country").ne(field("card_country")))
                .then("risk", 25))
            .add(Rule.named("velocity").when(field("tx_last_hour").ge(5)).then("risk", 30))
            .add(Rule.named("new_device_large").when(field("new_device").and(field("amount").gt(500)))
                .then("risk", 20))
            .add(Rule.named("disposable_email")
                .when(field("email").matches("@(mailinator|guerrillamail|10minutemail)\\.com$"))
                .then("risk", 35))
            .add(Rule.named("night_time").when(field("local_hour").between(0, 5)).then("risk", 10))
            .build();

        var tx = new Transaction(1890.0, 240.0, "NG", "ES", 2, true, "j.doe@mailinator.com", 14);
        var decision = flags.evaluate(tx);
        System.out.println(decision.fired());
        System.out.println(decision.outputs());
    }
}

// Design preview: the Go binding does not exist yet (phase H).
package main

import (
	"fmt"
	"log"

	cr "github.com/cleanrules/cleanrules/bindings/go"
)

type Transaction struct {
	Amount      float64 `json:"amount"`
	AvgAmount   float64 `json:"avg_amount"`
	IPCountry   string  `json:"ip_country"`
	CardCountry string  `json:"card_country"`
	TxLastHour  int     `json:"tx_last_hour"`
	NewDevice   bool    `json:"new_device"`
	Email       string  `json:"email"`
	LocalHour   int     `json:"local_hour"`
}

func main() {
	risk := func(name string, when cr.Expr, points int) *cr.Rule {
		return cr.NewRule(name, when).Then("risk", points)
	}
	flags, err := cr.NewRuleSet([]*cr.Rule{
		risk("large_amount", cr.Field("amount").Gt(cr.Field("avg_amount").Mul(3)), 40),
		risk("country_mismatch", cr.Field("ip_country").Ne(cr.Field("card_country")), 25),
		risk("velocity", cr.Field("tx_last_hour").Ge(5), 30),
		risk("new_device_large", cr.And(cr.Field("new_device"), cr.Field("amount").Gt(500)), 20),
		risk("disposable_email",
			cr.Field("email").Matches(`@(mailinator|guerrillamail|10minutemail)\.com$`), 35),
		risk("night_time", cr.Field("local_hour").Between(0, 5), 10),
	}, cr.WithHitPolicy(cr.RuleOrder))
	if err != nil {
		log.Fatal(err)
	}

	decision, err := flags.Evaluate(Transaction{
		Amount: 1890.0, AvgAmount: 240.0, IPCountry: "NG", CardCountry: "ES",
		TxLastHour: 2, NewDevice: true, Email: "j.doe@mailinator.com", LocalHour: 14,
	})
	if err != nil {
		log.Fatal(err)
	}
	fmt.Println(decision.Fired())
	fmt.Println(decision.Outputs())
}

// Design preview: the C binding does not exist yet (phase E).
#include <stdio.h>
#include <cleanrules.h>

static int fail(cr_status status) {
    fprintf(stderr, "%s: %s\n", cr_status_name(status), cr_error_message(cr_last_error()));
    return 1;
}

int main(void) {
    cr_ruleset *rules = NULL;
    cr_decision *decision = NULL;
    char *outputs = NULL;

    /* The rule set document shown in the JSON tab. */
    cr_status status = cr_ruleset_read_file("fraud.ruleset.json", &rules);
    if (status != CR_OK) return fail(status);

    const char *tx =
        "{\"amount\": 1890.0, \"avg_amount\": 240.0, \"ip_country\": \"NG\", "
        "\"card_country\": \"ES\", \"tx_last_hour\": 2, \"new_device\": true, "
        "\"email\": \"j.doe@mailinator.com\", \"local_hour\": 14}";
    status = cr_ruleset_evaluate_json(rules, tx, &decision);
    if (status == CR_OK) status = cr_decision_outputs_json(decision, &outputs);
    if (status == CR_OK) puts(outputs);  /* a JSON array: one payload per selected rule */
    else fail(status);

    cr_string_free(outputs);
    cr_decision_free(decision);
    cr_ruleset_free(rules);
    return status == CR_OK ? 0 : 1;
}

The decision document records every rule's result tree, with each value read; its outputs are derived from it. It equals decision.to_dict() in Python.

The explanation

Why, and why not.

Asking a rule_order decision for a single output raises HitPolicyError, with a hint. To get a total risk score, reuse the very same rules under collect with the sum aggregate: 40 + 25 + 20 + 35 = 120.
