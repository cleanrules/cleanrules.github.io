PricingANY

Discount agreement

A software company offers an education plan. Customers qualify through any of three channels: a student ID, an academic email address, or being under 26. All three must give the same offer; if someone later edits one channel, the inconsistency must not reach customers.

The problem

Discount agreement, in words.

Three rules with the same payload: the education plan at 50 % off. The any hit policy selects the first rule that fired and checks that every rule that fired has an equal payload, raising HitPolicyError if not.

input.json
{"student_id": "S-1188", "email": "ana@uni.edu", "age": 23}

A 23-year-old customer with a student ID and an address at uni.edu. The input in the JSON format, where decimals and dates have their own encoding; in Python it is the dict in the program below.

The rules in every language

One rule set, ten ways to write it.

Python and Rust run today, and the JSON tab is the rule set document the core writes. The other tabs are design previews of the planned bindings: they show the intended API, and the result they would print is the one the core computes for the same rules and data.

discounts.rsRust core · runs
//! Discount agreement: the ANY hit policy checks that rules agree.
use cleanrules::Value;
use cleanrules::build::{Build, constant, field};
use cleanrules::explain::explain;
use cleanrules::format::{ValueContext, read_value, write_value};
use cleanrules::json::{Limits, parse, to_string};
use cleanrules::rules::{Config, HitPolicy, Rule, RuleSet};

type Result<T> = std::result::Result<T, Box<dyn std::error::Error>>;

fn rule(name: &str, when: Build, then: Vec<(&str, Build)>) -> Result<Rule> {
    let then = then
        .into_iter()
        .map(|(key, value)| Ok((key.to_owned(), value.build()?)))
        .collect::<Result<Vec<_>>>()?;
    Ok(Rule::with_payload(name, when.build()?, then)?)
}

fn input(json: &str) -> Result<Value> {
    Ok(read_value(
        &parse(json, &Limits::default())?,
        "",
        ValueContext::Data,
    )?)
}

fn channels() -> Vec<(&'static str, Build)> {
    vec![
        ("student_id", !field("student_id").is_null()),
        (
            "academic_email",
            field("email").ends_with(".edu") | field("email").ends_with(".ac.uk"),
        ),
        ("under_26", field("age").lt(26)),
    ]
}

fn education_rules() -> Result<Vec<Rule>> {
    let education = || {
        vec![
            ("plan", constant("education")),
            ("discount_pct", constant(50)),
        ]
    };
    channels()
        .into_iter()
        .map(|(name, when)| rule(name, when, education()))
        .collect()
}

fn main() -> Result<()> {
    let rules = education_rules()?;
    let any = Config {
        hit_policy: HitPolicy::Any,
        ..Config::default()
    };
    let ruleset = RuleSet::new(rules.clone(), any)?;

    let customer = input(r#"{"student_id": "S-1188", "email": "ana@uni.edu", "age": 23}"#)?;
    let decision = ruleset.evaluate(&customer)?;
    if let Some(output) = decision.output()? {
        println!("output: {}", to_string(&write_value(&output)));
    }
    for entry in explain(&decision).rules {
        let note = entry
            .note
            .map(|n| format!("{} ({})", n.kind.as_str(), n.winner))
            .unwrap_or_default();
        println!("{:<15} {:<9} {note}", entry.rule, entry.status.as_str());
    }

    let drift = rule(
        "youth_promo",
        field("age").lt(26),
        vec![
            ("plan", constant("education")),
            ("discount_pct", constant(40)),
        ],
    )?;
    let drifted = RuleSet::new([rules, vec![drift]].concat(), any)?;
    match drifted.evaluate(&customer) {
        Ok(_) => println!("they agree"),
        Err(error) => println!("{}: {:?}", error.code(), error.context().rules),
    }
    Ok(())
}

// Design preview: the Java binding does not exist yet (phase G).
import io.github.cleanrules.*;
import java.util.Map;

import static io.github.cleanrules.Conditions.*;

public class Discounts {
    static Rule education(String name, Condition when) {
        return Rule.named(name).when(when).then("plan", "education").then("discount_pct", 50);
    }

    public static void main(String[] args) {
        var rules = RuleSet.builder()
            .hitPolicy(HitPolicy.ANY)
            .add(education("student_id", field("student_id").isNull().not()))
            .add(education("academic_email",
                field("email").endsWith(".edu").or(field("email").endsWith(".ac.uk"))))
            .add(education("under_26", field("age").lt(26)))
            .build();

        var customer = Map.of("student_id", "S-1188", "email", "ana@uni.edu", "age", 23);
        var decision = rules.evaluate(customer);
        System.out.println(decision.fired());
        System.out.println(decision.output().orElseThrow());
    }
}

discounts.godesign preview
// Design preview: the Go binding does not exist yet (phase H).
package main

import (
	"fmt"
	"log"

	cr "github.com/cleanrules/cleanrules/bindings/go"
)

func main() {
	education := func(name string, when cr.Expr) *cr.Rule {
		return cr.NewRule(name, when).Then("plan", "education").Then("discount_pct", 50)
	}
	email := cr.Field("email")
	rules, err := cr.NewRuleSet([]*cr.Rule{
		education("student_id", cr.Not(cr.Field("student_id").IsNull())),
		education("academic_email", cr.Or(email.EndsWith(".edu"), email.EndsWith(".ac.uk"))),
		education("under_26", cr.Field("age").Lt(26)),
	}, cr.WithHitPolicy(cr.Any))
	if err != nil {
		log.Fatal(err)
	}

	decision, err := rules.Evaluate(map[string]any{"student_id": "S-1188", "email": "ana@uni.edu", "age": 23})
	if err != nil {
		log.Fatal(err)
	}
	output, err := decision.Output()
	if err != nil {
		log.Fatal(err)
	}
	fmt.Println(decision.Fired())
	fmt.Println(output)
}

// Design preview: the C binding does not exist yet (phase E).
#include <stdio.h>
#include <cleanrules.h>

static int fail(cr_status status) {
    fprintf(stderr, "%s: %s\n", cr_status_name(status), cr_error_message(cr_last_error()));
    return 1;
}

int main(void) {
    cr_ruleset *rules = NULL;
    cr_decision *decision = NULL;
    char *output = NULL;

    /* The rule set document shown in the JSON tab. */
    cr_status status = cr_ruleset_read_file("discounts.ruleset.json", &rules);
    if (status != CR_OK) return fail(status);

    const char *customer =
        "{\"student_id\": \"S-1188\", \"email\": \"ana@uni.edu\", \"age\": 23}";
    status = cr_ruleset_evaluate_json(rules, customer, &decision);
    if (status == CR_OK) status = cr_decision_output_json(decision, &output);
    if (status == CR_OK) puts(output);  /* a JSON object, or null */
    else fail(status);

    cr_string_free(output);
    cr_decision_free(decision);
    cr_ruleset_free(rules);
    return status == CR_OK ? 0 : 1;
}

int main() {
    const cr::Value education = {{"plan", "education"}, {"discount_pct", 50}};

    std::cout << '\n';
    if (auto output = decision.output()) std::cout << output->to_json() << '\n';
}

The explanation

Why, and why not.

The first rule that fired is selected; the other two carry the note any_agrees: they fired, agreed, and the policy reports the first.

What happens if…

Change the input, see the outcome move.

Configuration drift. A new youth_promo rule gives the same plan at 40 %. For a customer it applies to, the outputs disagree and evaluation raises, listing every rule that fired. The mistake is caught at the first affected customer instead of being resolved silently by rule order.

A customer for whom only agreeing rules fire still gets the offer: a 30-year-old with an academic email matches academic_email alone.

from cleanrules import HitPolicyError

drift = Rule("youth_promo", F.age < 26, {"plan": "education", "discount_pct": 40})
try:
    RuleSet([*channels, drift], hit_policy="any").evaluate(customer)
except HitPolicyError as err:
    print(err.code, err.rules)
    print(err)

print(dict(RuleSet([*channels, drift], hit_policy="any").evaluate({"age": 30, "email": "bo@uni.edu"}).output))
